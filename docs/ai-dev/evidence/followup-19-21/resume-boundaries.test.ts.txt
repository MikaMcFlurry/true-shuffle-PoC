import {expect,it} from '/workspace/ts-review-19-21/node_modules/vitest/dist/index.js';
import {onboarded} from '/workspace/ts-review-19-21/test/hub/harness.ts';
import {RequestBudget} from '/workspace/ts-review-19-21/src/worker/spotify/client.ts';
const observe=(h:any)=>h.hub.sync(new RequestBudget(25),{force:true});
async function setup(){const h=await onboarded({tracks:200,durationMs:180000}); const id=h.stationIds[0]!;await h.hub.play(id);await h.hub.playerAction('pause');const s=h.hub.savedSession(id)!;const ids=JSON.parse(h.sql.first('SELECT deck FROM stations WHERE id = ?',id).deck).items.map((it:any)=>it.id);const hub:any=h.hub;const song=ids[120];const track=hub.lookupTracks([song],hub.usedSources()).get(song);return{h,hub,id,s,ids,song,track};}
it('legacy seen entry has no ten-minute extension after restart',async()=>{const{h,hub,s,song}=await setup();const at=h.clock.t;hub.kvSet('seen_plays',[{id:song,from:at-1000,to:at+60000}]);h.restart();h.clock.t+=4*60000;h.fake.user().recent.unshift({trackId:song,playedAt:h.clock.t,contextUri:s.contextUri});await observe(h);expect(h.sql.first('SELECT COUNT(*) AS n FROM plays WHERE track_id = ?',song).n).toBe(1);});
it('an open delayed first listing dedups once across restart, subsequent one survives',async()=>{const{h,hub,s,song,track}=await setup();const at=h.clock.t;hub.recordSeenPlay({id:song,at,start:at-45000,end:at,contextUri:s.contextUri,track,open:true});h.restart();h.clock.t+=4*60000;h.fake.user().recent.unshift({trackId:song,playedAt:h.clock.t,contextUri:s.contextUri});await observe(h);expect(h.sql.first('SELECT COUNT(*) AS n FROM plays WHERE track_id = ?',song).n).toBe(1);h.restart();h.clock.t+=3*60000;h.fake.user().recent.unshift({trackId:song,playedAt:h.clock.t,contextUri:s.contextUri});await observe(h);expect(h.sql.first('SELECT COUNT(*) AS n FROM plays WHERE track_id = ?',song).n).toBe(2);});
it('same-context new play after intervening heard song survives',async()=>{const{h,hub,s,ids,song,track}=await setup();const at=h.clock.t;hub.recordSeenPlay({id:song,at,start:at-45000,end:at,contextUri:s.contextUri,track,open:true});h.clock.t+=2*60000;h.fake.user().recent.unshift({trackId:ids[140],playedAt:h.clock.t,contextUri:s.contextUri});await observe(h);h.restart();h.clock.t+=2*60000;h.fake.user().recent.unshift({trackId:song,playedAt:h.clock.t,contextUri:s.contextUri});await observe(h);expect(h.sql.first('SELECT COUNT(*) AS n FROM plays WHERE track_id = ?',song).n).toBe(2);});
it('after extension deadline same track is a new play',async()=>{const{h,hub,s,song,track}=await setup();const at=h.clock.t;hub.recordSeenPlay({id:song,at,start:at-45000,end:at,contextUri:s.contextUri,track,open:true});h.clock.t+=600001;h.fake.user().recent.unshift({trackId:song,playedAt:h.clock.t,contextUri:s.contextUri});await observe(h);expect(h.sql.first('SELECT COUNT(*) AS n FROM plays WHERE track_id = ?',song).n).toBe(2);});
it('unknown progress and legacy heardSince never advance queue after restart',async()=>{const{h,id,s,ids}=await setup();h.sql.run('UPDATE playback_sessions SET data = ? WHERE station_id = ?',JSON.stringify({...s,progressMs:null,heardSince:h.clock.t}),id);h.sql.run('INSERT INTO plays (played_at,track_id,context_uri,station_id,ignored,meta) VALUES (?,?,?,?,0,NULL)',h.clock.t,ids[3],s.contextUri,id);h.restart();h.fake.user().player.contextUri='spotify:playlist:unrelated';expect(await h.hub.play(id)).toMatchObject({ok:true});expect(h.fake.current()).toBe(ids[0]);expect(h.fake.user().player.progressMs).toBe(0);expect(h.hub.savedSession(id)).toMatchObject({sessionId:s.sessionId,currentIndex:0});});
