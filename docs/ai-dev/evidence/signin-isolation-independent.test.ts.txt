import {it,expect} from 'vitest';
import fs from 'node:fs';
import ts from 'typescript';
import {SpotifyClient} from '../../src/worker/spotify/client';
import {SpotifyOperationGates} from '../../src/worker/spotify/operation-gates';
import {SpotifyUsageTracker} from '../../src/worker/spotify/usage';
import {sha256b64url} from '../../src/worker/lib/crypto';
import {nodeSql,T0} from '../hub/harness';
it('RT-T03 actual final index policy separates OAuth flows but counts prelogin together',async()=>{
 const gate=new SpotifyOperationGates(nodeSql()),tracker=new SpotifyUsageTracker(nodeSql(),()=>T0);let calls=0;
 const registry={spotifyOperationSnapshot:(id:string,op:string)=>gate.snapshot(id,op),spotifyOperationBlocked:(id:string,c:any)=>gate.block(id,c),finishSpotifyOperation:(id:string,op:string,rev:number)=>gate.finish(id,op,rev),recordSpotifyUsage:(id:string,m:any)=>tracker.record(id,m)};
 const source=fs.readFileSync(new URL('../../src/worker/index.ts',import.meta.url),'utf8');
 const start=source.indexOf('const flowKey = ');expect(start).toBeGreaterThan(-1);
 const end=source.indexOf('let tokens:',start);expect(end).toBeGreaterThan(start);
 const code=ts.transpileModule(source.slice(start,end),{compilerOptions:{target:ts.ScriptTarget.ES2022,module:ts.ModuleKind.None}}).outputText;
 const AsyncFunction=Object.getPrototypeOf(async function(){}).constructor;
 const factory=new AsyncFunction('registry','state','sha256b64url','boundedSpotifyRpc',code+'\nreturn spotifyPolicy;');
 const make=(state:string)=>factory(registry,state,sha256b64url,(value:any)=>Promise.resolve(value));
 const policyA=await make('private-flow-A'),policyB=await make('private-flow-B');
 const endpoint={accountsBase:'https://accounts.test',apiBase:'https://api.test/v1',clientId:'test-app'};
 const fetch=async()=>{calls++;return calls===1?new Response(JSON.stringify({error:{reason:'QUOTA_EXCEEDED'}}),{status:429,headers:{'Retry-After':'3600'}}):new Response(JSON.stringify({access_token:'synthetic',refresh_token:'synthetic',expires_in:3600}),{status:200});};
 const exchange=(code:string,policy:any)=>SpotifyClient.exchangeCode(endpoint,fetch,code,'https://app.test/callback','synthetic-verifier',T0,policy);
 await expect(exchange('code-A',policyA)).rejects.toMatchObject({status:429});
 await expect(exchange('code-B',policyB)).resolves.toMatchObject({accessToken:'synthetic'});
 await expect(exchange('code-A',policyA)).rejects.toMatchObject({status:429});
 expect(calls).toBe(2);
 const report=tracker.report(0);expect(report.observedListeners).toBe(0);expect(report.listeners.every(l=>l.listener==='Anmeldung')).toBe(true);
 expect(report.episodes[0].firstSuccessAt).toBeNull();expect(JSON.stringify(report)).not.toMatch(/private-flow|@signin|synthetic|code-A|code-B/);
});
