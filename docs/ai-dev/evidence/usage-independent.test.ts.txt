import {it,expect} from 'vitest';
import {SpotifyUsageTracker} from '../../src/worker/spotify/usage';
import {nodeSql,T0} from '../hub/harness';
const metric=(status:number)=>({endpoint:'/me/player/devices',method:'GET',category:'read' as const,status,retryCategory:status===429?'quota' as const:'none' as const,at:T0,retryAfter:status===429?'60':null,reason:status===429?'QUOTA_EXCEEDED':undefined});
it('RT-T01 another overflow listener cannot close failing listeners episode',()=>{
 const t=new SpotifyUsageTracker(nodeSql(),()=>T0);
 for(let i=1;i<=5;i++)t.record('listener-'+i,metric(200));
 t.record('listener-6',metric(429));t.record('listener-7',metric(200));
 const report=t.report(7);expect(report.episodes[0].firstSuccessAt).toBeNull();
 t.record('listener-6',metric(200));expect(t.report(7).episodes[0].firstSuccessAt).toBe(T0);
});
it('RT-T01 prelogin aggregate must not imply identified user recovery',()=>{
 const t=new SpotifyUsageTracker(nodeSql(),()=>T0);t.record('@signin',metric(429));t.record('@signin',metric(200));
 expect(t.report(0).episodes.every(e=>e.firstSuccessAt===null)).toBe(true);
});
it('separate prelogin actors cannot close each others quota episode',()=>{
 const t=new SpotifyUsageTracker(nodeSql(),()=>T0);t.record('@signin:FLOW_A',metric(429));t.record('@signin:FLOW_B',metric(200));
 expect(t.report(0).episodes[0].firstSuccessAt).toBeNull();t.record('@signin:FLOW_A',metric(200));expect(t.report(0).episodes[0].firstSuccessAt).toBe(T0);
 expect(JSON.stringify(t.report(0))).not.toMatch(/FLOW_A|FLOW_B|@signin/);
});
it('delayed response started before429 cannot close newer episode',()=>{
 const t=new SpotifyUsageTracker(nodeSql(),()=>T0);
 t.record('private',metric(429));t.record('private',{...metric(200),startedAt:T0-1});expect(t.report(1).episodes[0].firstSuccessAt).toBeNull();
 t.record('private',{...metric(200),startedAt:T0});expect(t.report(1).episodes[0].firstSuccessAt).toBe(T0);
});
it('old uncorrelated actorless success is cleared during additive migration',()=>{
 const sql=nodeSql();sql.run('CREATE TABLE spotify_usage_episodes(id INTEGER PRIMARY KEY AUTOINCREMENT, slot INTEGER NOT NULL, operation TEXT NOT NULL, first_failure_at INTEGER NOT NULL,last_failure_at INTEGER NOT NULL,reason TEXT,retry_after TEXT,earliest_retry_at INTEGER,attempts INTEGER NOT NULL,last_attempt_at INTEGER NOT NULL,last_status INTEGER NOT NULL,first_success_at INTEGER)');
 sql.run('INSERT INTO spotify_usage_episodes(slot,operation,first_failure_at,last_failure_at,attempts,last_attempt_at,last_status,first_success_at) VALUES(6,?,?,?,2,?,200,?)','GET /me/player/devices',T0,T0,T0,T0);
 const t=new SpotifyUsageTracker(sql,()=>T0);expect(t.report(7).episodes[0].firstSuccessAt).toBeNull();t.record('another-private',metric(200));expect(t.report(7).episodes[0].firstSuccessAt).toBeNull();
});
