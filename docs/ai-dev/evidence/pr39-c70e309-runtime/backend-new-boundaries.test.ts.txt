import { expect,it } from 'vitest';
import { LIKES_NAME } from '../../src/worker/hub/hub';
import { onboarded } from './harness';
import { RequestBudget } from '../../src/worker/spotify/client';
async function seed(){ const h=await onboarded({tracks:1800}); const ids=[...h.fake.tracks.keys()].slice(0,1700); h.sql.run(`INSERT INTO kv(k,v) VALUES('likes',?) ON CONFLICT(k) DO UPDATE SET v=excluded.v`,JSON.stringify({v:1,rev:2,items:ids.map(id=>({id,at:h.clock.t}))})); return {h,ids}; }
it.each(['POST','PUT'])('R132B-CONFIRMED: Like changes and retries preserve confirmed %s 120s cooldown',async(method)=>{const{h,ids}=await seed();let injected=false,at=0;h.fake.failWhen=(m,path)=>{const p=[...h.fake.playlists.values()].find(p=>p.name===LIKES_NAME);if(!injected&&p&&m===method&&path===`/playlists/${p.id}/items`&&(method==='PUT'||p.items.length===900)){injected=true;at=h.clock.t;return{status:429,headers:{'retry-after':'120'}};}return null;};await h.hub.retryLikesSync();await h.settle();await h.listen(1000);expect(injected).toBe(true);const calls=h.fake.calls.length;await h.hub.retryLikesSync();await h.hub.retryLikesSync();await h.hub.thumb([...h.fake.tracks.keys()][1750]!,1);await h.listen(60000);await h.settle();const actual=h.fake.calls.slice(calls).filter(c=>c.startsWith(method+' /v1/playlists/')&&c.includes('/items'));console.log('R132B-CONFIRMED',JSON.stringify({method,elapsed:h.clock.t-at,calls:actual,count:[...h.fake.playlists.values()].find(p=>p.name===LIKES_NAME)?.items.length,sync:(await h.hub.state()).likes?.sync}));expect(actual).toEqual([]);await h.listen(120000);for(let i=0;i<5;i++){await h.listen(60000);await h.settle();}expect([...h.fake.playlists.values()].find(p=>p.name===LIKES_NAME)!.items).toEqual([...ids,[...h.fake.tracks.keys()][1750]!]);},120000);
it('R132B-UNKNOWN: explicit retry after a 429 without Retry-After recovers exact Likes once provider answers',async()=>{const{h,ids}=await seed();let injected=false;h.fake.failWhen=(m,path)=>{const p=[...h.fake.playlists.values()].find(p=>p.name===LIKES_NAME);if(!injected&&p&&m==='POST'&&path===`/playlists/${p.id}/items`&&p.items.length===900){injected=true;return{status:429};}return null;};await h.hub.retryLikesSync();await h.settle();await h.listen(1000);expect(injected).toBe(true);const before=[...h.fake.playlists.values()].find(p=>p.name===LIKES_NAME)!.items.length;const calls=h.fake.calls.length;await h.hub.retryLikesSync();await h.settle();for(let i=0;i<5;i++){await h.listen(60000);await h.settle();}h.restart();await h.hub.retryLikesSync();await h.settle();await h.hub.retryLikesSync();await h.listen(60000);const p=[...h.fake.playlists.values()].find(p=>p.name===LIKES_NAME)!;console.log('R132B-UNKNOWN',JSON.stringify({before,after:p.items.length,calls:h.fake.calls.slice(calls).filter(c=>/^(PUT|POST) \/v1\/playlists\/.+\/items/.test(c)),job:h.sql.first(`SELECT state,run_after,error FROM jobs WHERE key='likes-sync'`),sync:(await h.hub.state()).likes?.sync}));expect(p.items).toEqual(ids);expect((await h.hub.state()).likes?.sync).toMatchObject({pending:false,error:null});},120000);

it('R132B-DUE: a newer observed listener song cancels a due automatic continuation',async()=>{const h=await onboarded({tracks:12,playlists:[6,6]});const sid=h.stationIds[0]!;await h.hub.play(sid);await h.listen(20000);await h.hub.sync(new RequestBudget(8),{force:true});const session=h.hub.savedSession(sid)!;const deck=h.fake.playlists.get(session.contextUri.split(':')[2]!)!;const outside=[...h.fake.tracks.keys()].filter(id=>!deck.items.includes(id));const b=outside[0]!,c=outside[1]!;h.fake.playSong('mika',b);h.fake.user('mika').player.progressMs=60000;await h.hub.sync(new RequestBudget(8),{force:true});await h.hub.continueAfter(b);await h.listen(h.fake.tracks.get(b)!.durationMs-60000-5000);h.clock.t+=5000;const p=h.fake.user('mika').player;p.isPlaying=false;p.progressMs=h.fake.tracks.get(b)!.durationMs;await h.hub.sync(new RequestBudget(8),{force:true});expect((await h.hub.state()).continueAfter?.due).toBe(true);h.clock.t+=100;h.fake.playSong('mika',c);await h.hub.sync(new RequestBudget(8),{force:true});const before=await h.hub.state();expect(before.observed?.track?.id).toBe(c);expect(before.observed?.state).toBe('playing');const calls=h.fake.calls.length;await h.listen(5000);const plays=h.fake.calls.slice(calls).filter(x=>x.startsWith('PUT /v1/me/player/play'));console.log('R132B-DUE',JSON.stringify({c,before:before.continueAfter,plays,after:h.fake.current(),result:(await h.hub.state()).continueAfter}));expect(plays).toEqual([]);expect(h.fake.current()).toBe(c);});
