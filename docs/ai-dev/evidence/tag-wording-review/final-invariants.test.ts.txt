import { describe, it, expect } from '/workspace/ts-review-4b0a/node_modules/vitest/dist/index.js';
import { onboarded } from '/workspace/ts-review-4b0a/test/hub/harness.ts';
import { MINUTE_MS } from '/workspace/ts-review-4b0a/src/core/types.ts';
function snapshot(h:any) {
 const tables=h.sql.all("SELECT name FROM sqlite_schema WHERE type='table' ORDER BY name");
 return JSON.stringify(tables.map((t:any)=>[t.name,h.sql.all('SELECT * FROM "'+t.name.replaceAll('"','""')+'"').map((r:any)=>JSON.stringify(r)).sort()]));
}
function identity(rows:any[]) {return rows.map(({facts,...rest})=>rest)}
describe('Independent temporal and read isolation attacks',()=>{
 it('cold and warm paginated reads preserve every SQLite table, calls, alarm and durable session',async()=>{
 const h=await onboarded({tracks:200,playlists:[100,100]}); const sid=h.stationIds[0]!;
 expect((await h.hub.play(sid)).ok).toBe(true);await h.listen(45*MINUTE_MS);
 const session=h.hub.savedSession(sid); const original=h.hub.history(200); expect(original.length).toBeGreaterThan(5);
 for(const cold of [true,false]) {
 if(cold)h.restart();
 const before=snapshot(h), calls=[...h.fake.calls],alarm=h.alarmAt();
 const rows=h.hub.history(200); const small=h.hub.history(3);const next=h.hub.history(3,small.at(-1)!.playedAt);
 expect(rows).toEqual(original);expect(next.every(e=>e.playedAt<small.at(-1)!.playedAt)).toBe(true);
 expect(rows.every(e=>e.facts!.kind===null)).toBe(true);
 expect(snapshot(h)).toBe(before);expect(h.fake.calls).toEqual(calls);expect(h.alarmAt()).toBe(alarm);expect(h.hub.savedSession(sid)).toEqual(session);
 }
 });
 it('later discovery record and deck replacement cannot relabel old plays or inflate their counts',async()=>{
 const h=await onboarded({tracks:200,playlists:[100,100]});const sid=h.stationIds[0]!;
 expect((await h.hub.play(sid)).ok).toBe(true);await h.listen(20*MINUTE_MS);
 const before=h.hub.history(200);expect(before.length).toBeGreaterThan(0);
 for(const row of before)h.sql.run("INSERT OR IGNORE INTO discoveries (station_id,id,source,score,status,meta,heard,created_at,updated_at) VALUES (?,?,'ai',1,'kept','[]',1,?,?)",sid,row.id,h.clock.t,h.clock.t);
 h.restart();const discovered=h.hub.history(200);expect(identity(discovered)).toEqual(identity(before));expect(discovered.map(e=>e.facts)).toEqual(before.map(e=>e.facts));
 expect((await h.hub.play(sid,null,{newQueue:true})).ok).toBe(true);h.restart();
 const replaced=h.hub.history(200);expect(identity(replaced)).toEqual(identity(before));expect(replaced.map(e=>e.facts)).toEqual(before.map(e=>e.facts));
 const snapshotBefore=snapshot(h);h.hub.history(200);expect(snapshot(h)).toBe(snapshotBefore);
 });
});
