import {describe,it,expect} from '<REVIEW_CHECKOUT>/node_modules/vitest/dist/index.js';
import {onboarded} from '<REVIEW_CHECKOUT>/test/hub/harness.ts';
import {HOUR_MS,MINUTE_MS} from '<REVIEW_CHECKOUT>/src/core/types.ts';
describe('Reachable history provenance',()=>{
 it('an actual fresh play of a kept song does not turn into a discovery after explicit replacement',async()=>{
 const h=await onboarded({tracks:200,playlists:[100,100]}); const sid=h.stationIds[0]!;await h.hub.play(sid);const id=h.hub.sessionView(sid)!.queue[0]!.track.id;
 // A previously kept recommendation is already part of a source playlist.
 h.sql.run("INSERT INTO discoveries (station_id,id,source,score,status,meta,heard,created_at,updated_at) VALUES (?,?,'ai',1,'kept','[]',2,?,?)",sid,id,h.clock.t-30*HOUR_MS,h.clock.t-30*HOUR_MS);
 await h.listen(20*MINUTE_MS);const old=h.hub.history(50).find(e=>e.id===id)!;expect(old).toBeDefined();expect(old.facts!.kind).toBe('fresh');
 h.clock.t+=25*HOUR_MS;expect(await h.hub.play(sid,null,{newQueue:true})).toMatchObject({ok:true});const after=h.hub.history(50).find(e=>e.id===id&&e.playedAt===old.playedAt)!;console.log('actual play changed',old.facts,after.facts);expect(after.facts!.kind).not.toBe('discovery');
 });
});
