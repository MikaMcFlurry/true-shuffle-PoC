import {describe,it,expect} from '<REVIEW_CHECKOUT>/node_modules/vitest/dist/index.js';
import {onboarded} from '<REVIEW_CHECKOUT>/test/hub/harness.ts';
import {DAY_MS,MINUTE_MS} from '<REVIEW_CHECKOUT>/src/core/types.ts';
describe('Fresh factual-history invariants',()=>{
 it('imported counts merge with memory while ignored guest rows do not count',async()=>{const h=await onboarded({tracks:200,playlists:[100,100]});const sid=h.stationIds[0]!;await h.hub.play(sid);const id=h.hub.sessionView(sid)!.queue[0]!.track.id,t=h.clock.t;h.sql.run('INSERT INTO memory (id,last_played_at,plays,early_skips,last_skipped_at,thumb) VALUES (?,?,2,0,NULL,0)',id,t-2*DAY_MS);h.sql.run('INSERT INTO plays (played_at,track_id,context_uri,station_id,ignored,meta) VALUES (?,?,NULL,?,1,NULL)',t,id,sid);h.hub.importHistory([[id,7,0,t-30*DAY_MS]],0,1);h.restart();const e=h.hub.history(1)[0]!;expect(e.ignored).toBe(true);expect(e.facts).toMatchObject({plays:9,lastPlayedAt:t-2*DAY_MS,kind:null});});
 it('actual guest playback is displayed with zero owner count',async()=>{const h=await onboarded({tracks:200,playlists:[100,100]});await h.hub.setGuest(true,2);await h.hub.play(h.stationIds[0]!);await h.listen(20*MINUTE_MS);const es=h.hub.history(50);expect(es.length).toBeGreaterThan(0);expect(es.every(e=>e.ignored)).toBe(true);expect(es.every(e=>e.facts?.plays===0)).toBe(true);});
 it('a history read during a genuine history 429 makes no provider request or cooldown mutation',async()=>{const h=await onboarded({tracks:200,playlists:[100,100]});await h.hub.play(h.stationIds[0]!);await h.listen(20*MINUTE_MS);const client=(h.hub as any).client({take(){},left:10});h.fake.failNext={status:429,count:1,body:{error:{reason:'QUOTA_EXCEEDED'}},headers:{'Retry-After':'300'}};await expect(client.recentlyPlayed()).rejects.toMatchObject({status:429});const before=h.fake.calls.length,kvs=h.sql.all('SELECT * FROM kv ORDER BY k'),alarm=h.alarmAt();expect(h.hub.history(50).length).toBeGreaterThan(0);expect(h.fake.calls.length).toBe(before);expect(h.sql.all('SELECT * FROM kv ORDER BY k')).toEqual(kvs);expect(h.alarmAt()).toBe(alarm);});
});
