import { describe,it,expect } from '<REVIEW_CHECKOUT>/node_modules/vitest/dist/index.js';
import { onboarded } from '<REVIEW_CHECKOUT>/test/hub/harness.ts';
import { DAY_MS, MINUTE_MS } from '<REVIEW_CHECKOUT>/src/core/types.ts';
import { songTags } from '<REVIEW_CHECKOUT>/src/client/components/song-tags-text.ts';
type H=Awaited<ReturnType<typeof onboarded>>;
async function fixture(){const h=await onboarded({tracks:200,playlists:[100,100]}); await h.hub.play(h.stationIds[0]!); const id=h.hub.sessionView(h.stationIds[0]!)!.queue[0]!.track.id;return {h,id,sid:h.stationIds[0]!};}
function add(h:H,id:string,sid:number|null,at:number){h.sql.run('INSERT INTO plays (played_at,track_id,context_uri,station_id,ignored,meta) VALUES (?,?,NULL,?,0,NULL)',at,id,sid);}
function deck(h:H,sid:number,id:string,items:any[]){const d=JSON.parse(h.sql.first<{deck:string}>('SELECT deck FROM stations WHERE id=?',sid)!.deck); d.items=items.map(x=>({id,state:'played',...x}));h.sql.run('UPDATE stations SET deck=? WHERE id=?',JSON.stringify(d),sid);h.restart();}
function snapshot(h:H){return JSON.stringify(h.sql.all<{name:string}>("SELECT name FROM sqlite_master WHERE type='table' ORDER BY name").map(x=>[x.name,h.sql.all('SELECT * FROM '+x.name)]),(_,v)=>v instanceof Uint8Array?Array.from(v):v);}
describe('Fresh history provenance falsification',()=>{
 it('a recommendation created weeks after the play must not retroactively supply its lane',async()=>{const {h,id,sid}=await fixture();const at=h.clock.t-20*DAY_MS;add(h,id,sid,at);h.sql.run("INSERT INTO discoveries (station_id,id,source,score,status,meta,heard,created_at,updated_at) VALUES (?,?,'ai',1,'candidate','[]',0,?,?)",sid,id,h.clock.t,h.clock.t);h.restart();const e=h.hub.history(1)[0]!;console.log('retroactive',JSON.stringify(e.facts),songTags(null,e.facts!,0,true,false,true));expect(e.facts!.kind).toBeNull();});
 it('repeated track occurrences with different lanes must preserve each exact occurrence',async()=>{const {h,id,sid}=await fixture();const a=h.clock.t-60*MINUTE_MS,b=h.clock.t;add(h,id,sid,a);add(h,id,sid,b);deck(h,sid,id,[{kind:'fresh',at:a},{kind:'favorite',at:b}]);const es=h.hub.history(2);console.log('repeated',es.map(e=>[e.playedAt,e.facts!.kind]));expect(es[0]!.facts!.kind).toBe('favorite');expect(es[1]!.facts!.kind).toBe('fresh');});
 it('an old rewritten deck play must not borrow a new occurrence two hours later',async()=>{const {h,id,sid}=await fixture();const at=h.clock.t-120*MINUTE_MS;add(h,id,sid,at);deck(h,sid,id,[{kind:'favorite',at:h.clock.t}]);const e=h.hub.history(1)[0]!;console.log('rewritten',e.facts);expect(e.facts!.kind).toBeNull();});
 it('history reads cause zero durable writes, provider calls, alarm changes and session changes',async()=>{const {h,id,sid}=await fixture();add(h,id,sid,h.clock.t);h.restart();const before=snapshot(h),calls=h.fake.calls.length,alarm=h.alarmAt(),session=h.hub.savedSession(sid);const es=h.hub.history(200);expect(es.length).toBeGreaterThan(0);expect(snapshot(h)).toBe(before);expect(h.fake.calls.length).toBe(calls);expect(h.alarmAt()).toBe(alarm);expect(h.hub.savedSession(sid)).toEqual(session);});
 it('station provenance does not leak another station recommendation',async()=>{const {h,id,sid}=await fixture();add(h,id,sid,h.clock.t-20*DAY_MS);h.sql.run("INSERT INTO discoveries (station_id,id,source,score,status,meta,heard,created_at,updated_at) VALUES (?,?,'ai',1,'kept','[]',1,?,?)",h.stationIds[1]!,id,h.clock.t-25*DAY_MS,h.clock.t);h.restart();expect(h.hub.history(1)[0]!.facts!.kind).toBeNull();});
});
