import { describe, expect, it } from "vitest";
import type { HubDeps } from "../../src/worker/hub/hub";
import {
	RequestBudget,
	type SpotifyClient,
	type SpotifyCooldown,
	type SpotifyCooldownScope,
} from "../../src/worker/spotify/client";
import { SpotifyGate, type SpotifyGateState } from "../../src/worker/spotify/gate";
import { onboarded, T0 } from "../hub/harness";

function shared() {
	const states = new Map<SpotifyCooldownScope | undefined, SpotifyGateState>();
	const gate = (scope?: SpotifyCooldownScope) =>
		new SpotifyGate(
			{
				get: () => states.get(scope) ?? null,
				set: (s) => {
					states.set(scope, s);
				},
			},
			() => T0,
		);
	const tokens = new Map<SpotifyCooldownScope | undefined, string>();
	const deps: NonNullable<HubDeps["sharedSpotify"]> = {
		getCooldown: async (s) => gate(s).get(tokens.get(s)),
		getSnapshot: async (s) => gate(s).snapshot(tokens.get(s)),
		setCooldown: async (c) => gate(c.scope).block(c, tokens.get(c.scope)),
		beginRecheck: async (scope) => {
			const t = gate(scope).begin("probe");
			if (t) tokens.set(scope, t);
			return !!t;
		},
		finishRecheck: async (ok, scope) => {
			const t = tokens.get(scope);
			if (t) gate(scope).finish(t, ok);
			tokens.delete(scope);
		},
		quarantineLegacyGlobal: async (c, r) => {
			const quarantined = gate().quarantine(c, r, gate("legacy-catalog"));
			return {
				quarantined,
				globalRevision: gate().snapshot().revision,
				catalogRevision: gate("legacy-catalog").snapshot().revision,
			};
		},
	};
	return { gate, deps };
}
const legacy: SpotifyCooldown = {
	kind: "quota",
	reason: "QUOTA_EXCEEDED",
	observedAt: T0,
	retryAfter: "3600",
	until: T0 + 3600000,
};
const client = (h: { hub: unknown }) =>
	(h.hub as { client(b: RequestBudget): SpotifyClient }).client(new RequestBudget(10));

describe('independent owner trial adversarial review',()=>{
 it('coalesces concurrent trial, caches after restart and never repeats endpoint',async()=>{
  const s=shared(),h=await onboarded({tracks:30,sharedSpotify:s.deps});s.gate().block(legacy);
  const before=h.fake.calls.length;
  const results=await Promise.all(Array.from({length:12},()=>h.hub.testSpotifyAvailability()));
  expect(h.fake.calls.slice(before)).toEqual(['GET /v1/me/player/devices','GET /v1/me/player','GET /v1/me/player/recently-played']);
  expect(results.every(r=>r.controls==='untested'&&r.catalog==='held')).toBe(true);
  h.restart();expect((await h.hub.testSpotifyAvailability()).cached).toBe(true);expect(h.fake.calls.length-before).toBe(3);
 });
 it('lost CAS response is recoverable after restart without republishing old global',async()=>{
  const s=shared();const original=s.deps.quarantineLegacyGlobal!;let lose=true;
  s.deps.quarantineLegacyGlobal=async(c,r)=>{const result=await original(c,r);if(lose){lose=false;throw new Error('lost reply');}return result;};
  const h=await onboarded({tracks:30,sharedSpotify:s.deps});s.gate().block(legacy);
  const before=h.fake.calls.length;await expect(h.hub.testSpotifyAvailability()).rejects.toThrow('lost reply');
  expect(h.fake.calls.length).toBe(before);h.restart();await h.hub.testSpotifyAvailability();
  expect(s.gate().get()).toBeNull();expect(s.gate('legacy-catalog').get()?.until).toBe(legacy.until);
  await expect(client(h).pause()).rejects.toMatchObject({scope:'legacy-catalog'});
 });
 it('new global after CAS blocks all GETs and is never cleared by final result/cache',async()=>{
  const s=shared(),original=s.deps.quarantineLegacyGlobal!;
  s.deps.quarantineLegacyGlobal=async(c,r)=>{const out=await original(c,r);s.gate().block({...legacy,endpoint:'/me',until:null});return out;};
  const h=await onboarded({tracks:30,sharedSpotify:s.deps});s.gate().block(legacy);const before=h.fake.calls.length;
  const result=await h.hub.testSpotifyAvailability();expect(result.stopped).toBe(true);expect(result.outcomes.devices.state).toBe('failed');expect(h.fake.calls.length).toBe(before);
  expect((await h.hub.testSpotifyAvailability()).outcomes.devices.state).toBe('held');expect(s.gate().get()?.endpoint).toBe('/me');expect(h.fake.calls.length).toBe(before);
 });
 it.each(['PUT','POST','DELETE'])('catalog quarantine blocks %s even exact essential path',async method=>{
  const s=shared(),h=await onboarded({tracks:30,sharedSpotify:s.deps});s.gate('legacy-catalog').block({...legacy,scope:'legacy-catalog'});const before=h.fake.calls.length;
  await expect(client(h).request(method,'/me/player')).rejects.toMatchObject({scope:'legacy-catalog'});expect(h.fake.calls.length).toBe(before);
 });
 it.each(['/me','/me/player/devices/extra','/me/player/recently-played/extra','/artists/'+ 'A'.repeat(22)+'/albums'])('catalog blocks non-exempt GET %s',async path=>{
  const s=shared(),h=await onboarded({tracks:30,sharedSpotify:s.deps});s.gate('legacy-catalog').block({...legacy,scope:'legacy-catalog'});const before=h.fake.calls.length;
  await expect(client(h).request('GET',path)).rejects.toMatchObject({scope:'legacy-catalog'});expect(h.fake.calls.length).toBe(before);
 });
 it('CAS stale revision and active probe cannot move global',()=>{
  const s=shared();s.gate().block({...legacy,until:null});const c=s.gate().get()!;const r=s.gate().snapshot().revision;s.gate().begin('running');
  expect(s.gate().quarantine(c,r,s.gate('legacy-catalog'))).toBe(false);expect(s.gate('legacy-catalog').get()).toBeNull();
 });
});
